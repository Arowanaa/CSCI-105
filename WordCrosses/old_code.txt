string readWordList(const string& filename, int len) {

    random_device myEngine;
    mt19937 eng(myEngine());
    uniform_real_distribution<double> randomReal(1.0, 40.0);

    string word = "";
    string out = "";
    ifstream file(filename);

    int rand = (int)randomReal(eng);
    int count = 0;

    if (!file.is_open()) {
        return "null";
    } else {
        while (file >> word) {
            if (word.length() == len) {
                count++;
                if (count == rand) {
                    word = out;
                    break;
                    }
                }
        }
        file.close();
    }
    return out;
}
